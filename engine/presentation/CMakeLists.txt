include(FetchContent)
option(VICE_ENABLE_AUDIO_OUTPUT "Build native output behind the VICE audio boundary" ON)
add_library(vice_presentation STATIC src/assets.cpp src/system.cpp src/skinning.cpp)
add_library(VICE::Presentation ALIAS vice_presentation)
target_include_directories(vice_presentation PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
target_link_libraries(vice_presentation PUBLIC VICE::World VICE::Math)
target_compile_features(vice_presentation PUBLIC cxx_std_23)
vice_enable_warnings(vice_presentation)
if(VICE_ENABLE_AUDIO_OUTPUT)
  FetchContent_Declare(miniaudio GIT_REPOSITORY https://github.com/mackron/miniaudio GIT_TAG f40cf03f80cdb7e741d43e53b7e706e8c1394bcf SOURCE_SUBDIR vice-no-subproject)
  FetchContent_MakeAvailable(miniaudio)
  target_sources(vice_presentation PRIVATE src/device.cpp)
  target_include_directories(vice_presentation SYSTEM PRIVATE ${miniaudio_SOURCE_DIR})
  target_compile_definitions(vice_presentation PRIVATE VICE_NATIVE_AUDIO=1)
  if(APPLE)
    target_link_libraries(vice_presentation PRIVATE "-framework CoreAudio" "-framework AudioToolbox" "-framework CoreFoundation")
  elseif(WIN32)
    target_link_libraries(vice_presentation PRIVATE ole32 uuid)
  endif()
else()
  target_sources(vice_presentation PRIVATE src/device_null.cpp)
endif()
